% --- CÂU 1 (TRANG 1) ---
\begin{ex}
	Thực hiện phép tính:
	\begin{enumerate}[a)]
		\item $A = 3x^3y \cdot 4xy^2$
		\item $B = 3x^2y^3 \cdot 2x^2y - 11x^4y^4$
		\item $C = 18x^6y^5 : 3x^2y^2 - 12x^4y^3$
		\item $D = 4x^3y^2 + 8x^5y^3 : 2x^2y$
	\end{enumerate}
	\loigiai{
		\begin{enumerate}[a)]
			\item Ta có:
			\begin{align*}
				A &= 3x^3y \cdot 4xy^2 \\
				&= (3 \cdot 4) \cdot (x^3 \cdot x) \cdot (y \cdot y^2) \\
				&= 12x^4y^3
			\end{align*}
			\item Ta có:
			\begin{align*}
				B &= 3x^2y^3 \cdot 2x^2y - 11x^4y^4 \\
				&= 6x^4y^4 - 11x^4y^4 \\
				&= (6 - 11)x^4y^4 \\
				&= -5x^4y^4
			\end{align*}
			\item Ta có:
			\begin{align*}
				C &= 18x^6y^5 : 3x^2y^2 - 12x^4y^3 \\
				&= 6x^4y^3 - 12x^4y^3 \\
				&= (6 - 12)x^4y^3 \\
				&= -6x^4y^3
			\end{align*}
			\item Ta có:
			\begin{align*}
				D &= 4x^3y^2 + 8x^5y^3 : 2x^2y \\
				&= 4x^3y^2 + 4x^3y^2 \\
				&= 8x^3y^2
			\end{align*}
		\end{enumerate}
	}
\end{ex}
